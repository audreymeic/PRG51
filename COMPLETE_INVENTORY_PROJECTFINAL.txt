#include <iostream>
using namespace std;

int main() {

    int choice;
    string Item_name;
    int Price = 0, Quantity = 0;
    int found = 0;

    do {

        cout << "\n===== INVENTORY MENU =====\n";
        cout << "1. Add Item\n";
        cout << "2. View Item\n";
        cout << "3. Remove Item\n";
        cout << "4. Search Item\n";
        cout << "5. Exit\n";

        cout << "\nENTER CHOICE: ";
        cin >> choice;

        switch (choice) {

    
            case 1: {
                if (found == 0) {
                    cout << "\nEnter Item Name: ";
                    cin >> Item_name;

                    cout << "Enter Quantity: ";
                    cin >> Quantity;

                    cout << "Enter Price: ";
                    cin >> Price;

                    found = 1;

                    cout << "\nItem added successfully!";
                }
                else {
                    cout << "\nAn item already exists!";
                }

                break;
            }

            case 2: {
                if (found == 1) {
                    cout << "\n===== ITEM DETAILS =====";
                    cout << "\nItem Name: " << Item_name;
                    cout << "\nQuantity: " << Quantity;
                    cout << "\nPrice: " << Price;
                }
                else {
                    cout << "\nNo item available!";
                }

                break;
            }

            case 3: {
                if (found == 1) {
                    Item_name = "";
                    Quantity = 0;
                    Price = 0;
                    found = 0;

                    cout << "\nItem removed successfully!";
                }
                else {
                    cout << "\nNo item available!";
                }

                break;
            }

            case 4: {
                string search;

                cout << "\nEnter item name to search: ";
                cin >> search;

                if (found == 1 && Item_name == search) {
                    cout << "\nItem found!";
                    cout << "\nItem Name: " << Item_name;
                    cout << "\nQuantity: " << Quantity;
                    cout << "\nPrice: " << Price;
                }
                else {
                    cout << "\nItem not found!";
                }

                break;
            }

            case 5: {
                cout << "\nExiting Inventory System...";
                break;
            }

            default: {
                cout << "\nInvalid choice!";
                break;
            }
        }

    } while (choice != 5);

    return 0;
}
