#include <iostream>
using namespace std;

int main() {

    int choice;
    string Item_name;
    int Price = 0, Quantity = 0;
    int found = 0;

    do {

        cout << "\n===== INVENTORY MENU =====\n";
        cout << "1. Add Item\n";
        cout << "2. View Item\n";
        cout << "3. Remove Item\n";
        cout << "4. Search Item\n";
        cout << "5. Exit\n";

        cout << "\nENTER CHOICE: ";
        cin >> choice;

        switch (choice) {

           
            case 4: {
                string search;

                cout << "\nEnter item name to search: ";
                cin >> search;

                if (found == 1 && Item_name == search) {
                    cout << "\nItem found!";
                    cout << "\nItem Name: " << Item_name;
                    cout << "\nQuantity: " << Quantity;
                    cout << "\nPrice: " << Price;
                }
                else {
                    cout << "\nItem not found!";
                }

                break;
            }

        }

    } while (choice != 5);

    return 0;
}
